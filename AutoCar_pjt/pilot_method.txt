Pilot 메서드
Car = Pilot.AutoCar() # Car 객체 생성
Car.setSpeed(50) # 속도 설정 (0 ~ 99)
Car.steering       # 조향 -1 ~ 1
Car.forward() # 전진
Car.backward() # 후진
Car.camPan(n) # 카메라 좌우 (0 ~ 180)
Car.camTilt(n) # 카메라 상하 (-30 ~ 90)
Car.joystick() # 조이스틱
Car.getGyro(axis) # 자이로
Car.getAccel(axis) # 가속도
